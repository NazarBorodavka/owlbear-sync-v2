# Core vision (tracker/vision/{geometry,codebook,decode,detect,track,pipeline}.py)
# is deliberately pure numpy -- no native dependency, validated end-to-end
# against synthetic sequences in tests/. See docs/VALIDATION.md.
numpy>=1.24.0,<2.6

# Only the calibration package (tracker/vision/calibration/) and video I/O
# need OpenCV: chessboard intrinsic calibration, homography, perspective
# warp, and RTSP capture are all solved problems there, not worth
# reimplementing. Pinned rather than ranged because calibration output
# feeds a PHYSICAL setup -- an intrinsics change between versions would
# silently shift the mm-per-pixel scale that token positions depend on.
# This exact pin is what CI installs and runs the calibration tests
# against, so the pin and the validation stay in step.
#
# 4.13.0.92 specifically, matching the opencv-python the test suite was
# validated against locally. Do NOT drop this below 4.10: wheels built
# before then link numpy 1.x's ABI, and with numpy 2.x resolved above
# they fail at import with "numpy.core.multiarray failed to import".
# The Dockerfile's explicit `import cv2` check exists to catch exactly
# that class of breakage at build time rather than at runtime.
opencv-python-headless==4.13.0.92

Flask>=3.0,<3.1
Flask-HTTPAuth>=4.8,<4.9
